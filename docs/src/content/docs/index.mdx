```kiwi nocheck
# Count words in a file and print the top five.
counts = {}
for word in fio::read("README.md").lowercase().scan('[a-z]+') do
  counts[word] = counts.get(word, 0) + 1
end

top = counts.keys().sort(do (a, b) => counts[a] > counts[b]).take(5)
cli::table(["word", "count"], top.map(do (w) => [w, counts[w]]))
```
